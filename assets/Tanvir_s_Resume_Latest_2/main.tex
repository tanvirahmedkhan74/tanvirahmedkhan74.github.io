\researchEntry{ERL --- Evidence selection in multi-hop question answering}
{First author\enspace\textperiodcentered\enspace In preparation\statusbreak Planned target: ACL}
{\textit{From Transit Routing to Evidence Selection: Conditional Audits of Multi-Hop QA
Failures.}}
{https://tanvirahmedkhan74.github.io/research/erl/}
\begin{points}
\item Trace where supporting evidence is lost across candidate access, selection and reading,
using controlled studies rather than end-to-end accuracy alone.
\item Lead the research design, implementation and analysis. Selection gains are qualified
against stronger baselines and reader controls, and negative routing findings are retained.
\end{points}

\researchEntry{RIFT --- Reliability of renderer measurements}
{First author\enspace\textperiodcentered\enspace In preparation\statusbreak Planned target: TMLR}
{\textit{Finite-difference probes across hard feature selection: a frozen-index audit of one
released talking-head renderer.}}
{https://tanvirahmedkhan74.github.io/research/rift/}
\begin{points}
\item Hold feature selections fixed so that a measured difference separates a change of discrete
choice from the renderer's local response.
\item Lead the experimental design, implementation and interpretation within a single-renderer
scope, retaining negative and ambiguous findings.
\end{points}

\researchEntry{Deadline articulation --- Information available at playback}
{First author\enspace\textperiodcentered\enspace Work in progress\statusbreak
Planned: NeurIPS 2027 / ICCV 2027}
{\textit{Deadline-Available Speech States for Full-Duplex Avatar Articulation.}}
{https://tanvirahmedkhan74.github.io/research/deadline-articulation/}
\begin{points}
\item Examine what articulation-relevant information streaming speech-model states carry at a
playback deadline, and what additional waiting contributes.
\item A completed pilot diagnostic motivates an expanded study of trained readouts and timing
schedules; that expanded programme remains in development.
\end{points}
