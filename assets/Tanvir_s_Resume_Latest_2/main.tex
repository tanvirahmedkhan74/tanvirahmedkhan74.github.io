\section{First-Author Manuscripts in Preparation}
\study{ERL: Evidence selection in multi-hop question answering}
{First author; manuscript in preparation. Planned target: ACL.}
{\textit{From Transit Routing to Evidence Selection: Conditional Audits of Multi-Hop QA Failures.}
Design controlled studies tracing support loss through evidence access, selection and reading.
The work tests a routing hypothesis, qualifies stronger selection baselines and examines when
readers use the supplied support.}
{https://tanvirahmedkhan74.github.io/research/erl/}

\study{RIFT: Finite-difference probes across hard feature selection}
{First author; manuscript in preparation. Planned target: TMLR.}
{\textit{A frozen-index audit of one released talking-head renderer.}
Study when finite-difference probes reflect local response as discrete feature choices change.
Fixed-selection controls separate measurement effects from the renderer's response, retaining
negative and ambiguous findings within the one-renderer scope.}
{https://tanvirahmedkhan74.github.io/research/rift/}

\study{Deadline-Available Speech States for Full-Duplex Avatar Articulation}
{First author; work in progress. Alternative planned targets: NeurIPS 2027 or ICCV 2027.}
{Study what articulation-relevant information streaming speech-model states contain at a
playback deadline, and what additional waiting contributes. A completed pilot diagnostic
motivates an expanded study of trained readouts and timing schedules; the expanded programme
remains in development.}
{https://tanvirahmedkhan74.github.io/research/deadline-articulation/}

\section{Independent Research in Progress}
